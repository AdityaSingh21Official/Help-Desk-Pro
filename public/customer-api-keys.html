/ JavaScript / Node.js 

const response = await fetch("https://your-domain.com/api/tickets", {
  method: "POST",
  headers: {
    "Authorization": "Bearer hd_live_xxxxxxxx",
    "Content-Type": "application/json"
  },
  body: JSON.stringify({
    subject: "Issue",
    description: "Details"
  })
});

const data = await response.json();

/ Python (requests) 

import requests

response = requests.post(
    "https://your-domain.com/api/tickets",
    headers={"Authorization": "Bearer hd_live_xxxxxxxx"},
    json={"subject": "Issue", "description": "Details"},
)

print(response.json())

/ Replace hd_live_xxxxxxxx with your key. Keep it secret and never put it in public code.
